class Shape2D {
    public int length, breadth, height, base, side1, side2, side3;
    public String type;

    public Shape2D() { // Square [cite: 12]
        this.length = 5;
        this.type = "Square";
        System.out.println("A Square has been created with length: " + length);
    }

    public Shape2D(int l, int b) { // Rectangle [cite: 12]
        this.length = l;
        this.breadth = b;
        this.type = "Rectangle";
        System.out.println("A Rectangle has been created with length: " + length + " and breadth: " + breadth);
    }

    public Shape2D(int h, int b, String t) { // Triangle Type 1 [cite: 12]
        this.height = h;
        this.base = b;
        this.type = t;
        System.out.println("A Triangle has been created with height: " + height + " and base: " + base);
    }

    public Shape2D(int s1, int s2, int s3) { // Triangle Type 2 [cite: 14]
        this.side1 = s1; this.side2 = s2; this.side3 = s3;
        this.type = "Triangle";
        System.out.println("A Triangle has been created with the following sides: " + s1 + ", " + s2 + ", " + s3);
    }

    public void area() {
        if (type.equals("Square")) System.out.println("The area of the Square is:\n" + (double)(length * length));
        else if (type.equals("Rectangle")) System.out.println("The area of the Rectangle is:\n" + (double)(length * breadth));
        else if (height != 0) System.out.println("The area of the Triangle is:\n" + (0.5 * height * base));
        else {
            double s = (side1 + side2 + side3) / 2.0;
            double area = Math.sqrt(s * (s - side1) * (s - side2) * (s - side3));
            System.out.printf("The area of the Triangle is:\n%.2f\n", area); [cite: 16]
        }
    }
}

class Triangle {
    public int s1, s2, s3;
    public Triangle(int a, int b, int c) { s1 = a; s2 = b; s3 = c; }
    
    public void triangleDetails() {
        System.out.println("Three sides of the triangle are: " + s1 + ", " + s2 + ", " + s3);
        System.out.println("Perimeter: " + (s1 + s2 + s3));
    }

    public String printTriangleType() {
        if (s1 == s2 && s2 == s3) return "This is an Equilateral Triangle."; [cite: 29]
        if (s1 == s2 || s2 == s3 || s1 == s3) return "This is an Isosceles Triangle."; [cite: 30]
        return "This is a Scalene Triangle."; [cite: 31]
    }

    public void compareTriangles(Triangle t) {
        if (this == t) System.out.println("These two triangle objects have the same address."); [cite: 33]
        else if (s1 == t.s1 && s2 == t.s2 && s3 == t.s3) System.out.println("Addresses are different but the sides of the triangles are equal."); [cite: 33]
        else if ((s1+s2+s3) == (t.s1+t.s2+t.s3)) System.out.println("Only the perimeter of both triangles is equal."); [cite: 33]
        else System.out.println("Addresses, length of the sides and perimeter all are different."); [cite: 33]
    }
}

    public void board() { System.out.println("No passengers"); }
    public void board(NubStudent s) {
        if (!s.pass) System.out.println("You don't have a bus pass!"); [cite: 74]
        else if (!s.home.equals(route)) System.out.println("You got on the wrong bus!"); [cite: 74]
        else if (count >= max) System.out.println("Bus is full!"); [cite: 74]
        else { passengers[count++] = s; System.out.println(s.name + " boarded the bus."); }
    }
    public void board(NubStudent s1, NubStudent s2) { board(s1); board(s2); }
    public void showDetails() {
        System.out.println("Bus Route: " + route + "\nPassenger Count: " + count + " (Max: " + max + ")\nPassengers on Board:");
        for (int i = 0; i < count; i++) System.out.print(passengers[i].name + " ");
        System.out.println();
    }
